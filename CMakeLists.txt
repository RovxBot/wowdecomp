        "In-source builds not allowed.

        You may need to remove CMakeCache.txt."

endif ()

option(WOWDECOMP_BUILD_NATIVE_FRAMESCRIPT "Build original FrameScript ownership/events (no FrameXML/login)" OFF)

# Off by default: the client checks MPQ file MD5s (signaturefile) and GlueXML/FrameXML/AddOn .toc.sig files like the
# original. ON accepts unsigned or modified game data and interface files.

if(WOWDECOMP_BUILD_NATIVE_FILES)

    endif()
    add_subdirectory(Engine/Source/PortableFiles)
endif()
if(WOWDECOMP_BUILD_NATIVE_ASSETS)
    if(NOT WOWDECOMP_BUILD_NATIVE_FILES)
        message(FATAL_ERROR "Native assets require the native filesystem/runtime.")
    endif()
    add_subdirectory(Engine/Source/PortableAssets)
endif()

if(WOWDECOMP_BUILD_NATIVE_ASYNC)
    if(NOT WOWDECOMP_BUILD_NATIVE_ASSETS)
        message(FATAL_ERROR "Native asynchronous files require native assets/files/runtime.")
    endif()
    add_subdirectory(Engine/Source/PortableAsync)
endif()

if(WOWDECOMP_BUILD_NATIVE_WINDOW)
    if(NOT WOWDECOMP_BUILD_NATIVE_ASYNC)
        message(FATAL_ERROR "Native window/shell requires native asynchronous file services.")
    endif()
    add_subdirectory(Engine/Source/PortableWindow)
endif()
if(WOWDECOMP_BUILD_NATIVE_GL)
    add_subdirectory(Engine/Source/PortableGx)
endif()
if(WOWDECOMP_BUILD_NATIVE_PACKETS)
    if(NOT WOWDECOMP_BUILD_PORTABLE_CORE)
        message(FATAL_ERROR "Native packet services require the portable core runtime.")
    endif()
    add_subdirectory(Engine/Source/PortablePackets)
endif()
if(WOWDECOMP_BUILD_NATIVE_LUA)
    if(NOT WOWDECOMP_BUILD_PORTABLE_CORE)
        message(FATAL_ERROR "Native Lua requires the portable core runtime.")
    endif()
    add_subdirectory(Engine/Source/PortableLua)
endif()

if(WOWDECOMP_BUILD_NATIVE_FRAMESCRIPT)
    if(NOT WOWDECOMP_BUILD_NATIVE_LUA OR NOT WOWDECOMP_BUILD_NATIVE_ASYNC)
        message(FATAL_ERROR "Native FrameScript requires product Lua and retained file services.")
    endif()
    add_subdirectory(Engine/Source/PortableFrameScript)
endif()

if(WOWDECOMP_BUILD_NATIVE_XML)
    if(NOT WOWDECOMP_BUILD_NATIVE_FRAMESCRIPT)
        message(FATAL_ERROR "Native XML requires the FrameScript owner and retained source provider.")
    endif()
    add_subdirectory(Engine/Source/PortableXML)
endif()

if(WOWDECOMP_BUILD_NATIVE_LAYOUT)
    if(NOT WOWDECOMP_BUILD_NATIVE_XML)
        message(FATAL_ERROR "Native layout requires the XML and FrameScript owners.")
    endif()
    add_subdirectory(Engine/Source/PortableLayout)
endif()

if(WOWDECOMP_BUILD_NATIVE_FRAME_FONTS)
    if(NOT WOWDECOMP_BUILD_NATIVE_LAYOUT OR NOT WOWDECOMP_BUILD_NATIVE_GL)
        message(FATAL_ERROR "Native Frame fonts require layout/XML/VM and native Gxu/Gx owners.")
    endif()
    add_subdirectory(Engine/Source/PortableFrameFonts)
endif()

if(NOT WOWDECOMP_BUILD_CLIENT)
    if(NOT WOWDECOMP_BUILD_PORTABLE_CORE)
        message(FATAL_ERROR "Select at least one build target.")
    endif()
    return()
endif()
include(cmake/WindowsClient.cmake)

# Game identity (build 12340, version 3.3.5, client name) lives in GameVersion.h at the repository root.
# WaystoneVersion.h is regenerated from git on every build; the login screen shows its build number.
set(WAYSTONE_VERSION_HEADER ${CMAKE_BINARY_DIR}/generated/WaystoneVersion.h)
add_custom_target(WaystoneVersion
    COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=${CMAKE_SOURCE_DIR} -DOUTPUT=${WAYSTONE_VERSION_HEADER} -P ${CMAKE_SOURCE_DIR}/cmake/GenerateWaystoneVersion.cmake
    BYPRODUCTS ${WAYSTONE_VERSION_HEADER}
    VERBATIM
)
include_directories(${CMAKE_BINARY_DIR}/generated)
