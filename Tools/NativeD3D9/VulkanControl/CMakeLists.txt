cmake_minimum_required(VERSION 3.21)
project(WowVulkanLifetimeControl LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    message(FATAL_ERROR "This lifetime control requires Linux x86-64")
endif()
find_package(Vulkan REQUIRED)
option(CONTROL_SANITIZERS "Instrument the control with ASan/UBSan" OFF)
add_executable(VulkanLifetimeCheck main.cpp)
target_compile_features(VulkanLifetimeCheck PRIVATE cxx_std_17)
target_compile_options(VulkanLifetimeCheck PRIVATE -Wall -Wextra -Werror)
target_link_libraries(VulkanLifetimeCheck PRIVATE Vulkan::Vulkan)
if(CONTROL_SANITIZERS)
    target_compile_options(VulkanLifetimeCheck PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g1)
    target_link_options(VulkanLifetimeCheck PRIVATE -fsanitize=address,undefined -shared-libasan)
endif()
set_target_properties(VulkanLifetimeCheck PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN/lib")
